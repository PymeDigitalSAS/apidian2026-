#COMANDOS DE INSTALACION (usar la terminal de laragon):
git config --global http.sslverify false
git clone https://git.buho.la/facturalatam/co-apidian2026.git apidian2026
cd apidian2026
cp .env.example .env
#[Nota] Crear la base de datos en laragon
#[Nota] Abra el archivo .env y verifique la cadena de conexión a la base de datos, son las variables con prefijos DB_ y tambien agregar esta linea para la conversion del certificado URL_API_CERT_MODERNIZER="http://62.146.176.127:8091"
composer self-update 2.8.0
#[Nota] Puede ver la versión de composer instalada con: composer -V  (los Laragon antiguos traen composer 1.x incluido)
#[Nota] Si da error de permisos por haber instalado en la unidad C, abra visual studio como Administrador
#[Nota] CASO ESPECIAL: si el self-update falla con
#           rename(...composer-temp.phar, ...composer.phar): Acceso denegado (code: 5)
#       eso NO es un problema de permisos y ejecutar como Administrador NO lo soluciona.
#       Causa: Windows no permite sobrescribir composer.phar porque es el mismo archivo que está
#       corriendo el self-update en ese momento (y a veces el antivirus retiene el phar recién
#       descargado). Es típico del composer 1.x viejo que traían los Laragon antiguos; el
#       composer 2.x ya no sufre esto.
#       SOLUCIÓN: la versión nueva ya quedó descargada como composer-temp.phar.
#       En la carpeta C:\laragon\bin\composer ejecutar:
#           ren composer.phar composer-viejo.phar
#           ren composer-temp.phar composer.phar
#           composer -V    (debe mostrar 2.8.0; después puede borrar composer-viejo.phar)
#       Si composer-temp.phar no existiera, descargar el phar directamente (sin usar self-update):
#           php -r "copy('https://getcomposer.org/download/2.8.0/composer.phar','C:/laragon/bin/composer/composer.phar');"
#       Este error solo ocurre en Windows local; en Linux/VPS no existe.

composer install
#[nota] Si sale error por el paquete patie/laravel-backup, entonces debe activar la extensión zip desde laragon o desde php.ini ;extension=zip

php artisan key:generate
php artisan migrate --seed
php artisan migrate
unzip storage.zip
chmod -R 777 storage
chmod -R 777 bootstrap/cache
chmod -R 777 vendor/mpdf/mpdf
php artisan storage:link
#[nota] Las plantillas "urn" ya se aplican automáticamente con composer install/update.
#       Solo si necesita re-aplicarlas manualmente: php urn_on.php  (en PowerShell el .bat se ejecuta con .\urn_on.bat)

php artisan config:clear
php artisan cache:clear
php artisan config:cache
